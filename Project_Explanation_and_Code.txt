CERTIFYCHAIN: FULL PROJECT EXPLANATION AND CODEBASE
======================================================================
This project is a decentralized application (dApp) built to issue and verify certificates on a local Ethereum blockchain (Ganache).

Project Architecture:
1. Smart Contract: Written in Solidity, deployed to local Ganache. Manages storing Hashes.
2. Backend (Node.js/Express): Handles Tesseract OCR extraction, IPFS interactions (mocked), and blockchain queries via ethers.js.
3. Frontend (React/Vite): Contains the UI with a Dark Glassmorphism theme, Web3 wallet connection, and role-based portals (Admin/Student/Verifier).

Below is the complete codebase with file-by-file explanations.
======================================================================

FILE: smart_contracts\contracts\CertificateRegistry.sol

EXPLANATION: The Solidity smart contract. It maps file hashes to certificate data and ensures only the administrator can issue certificates.

======================================================================


// SPDX-License-Identifier: MIT
pragma solidity ^0.8.24;

contract CertificateRegistry {
    address public admin;

    struct Certificate {
        string certificateId;
        string studentName;
        string course;
        string ipfsHash;
        string fileHash;
        uint256 timestamp;
        bool isValid;
    }

    // Primary lookup: fileHash  → Certificate
    mapping(string => Certificate) private certsByHash;

    // Secondary lookup: certificateId → fileHash (for QR code / ID-based lookup)
    mapping(string => string) private hashByCertId;

    event CertificateIssued(string fileHash, string certificateId, string studentName);

    modifier onlyAdmin() {
        require(msg.sender == admin, "Only admin can perform this action");
        _;
    }

    constructor() {
        admin = msg.sender;
    }

    /// @notice Issue a certificate — only callable by admin
    function issueCertificate(
        string memory _fileHash,
        string memory _certificateId,
        string memory _studentName,
        string memory _course,
        string memory _ipfsHash
    ) public onlyAdmin {
        require(!certsByHash[_fileHash].isValid, "Certificate with this file hash already exists");
        require(bytes(hashByCertId[_certificateId]).length == 0, "Certificate ID already registered");

        certsByHash[_fileHash] = Certificate({
            certificateId: _certificateId,
            studentName: _studentName,
            course: _course,
            ipfsHash: _ipfsHash,
            fileHash: _fileHash,
            timestamp: block.timestamp,
            isValid: true
        });

        // Enable ID-based lookup (for QR verification)
        hashByCertId[_certificateId] = _fileHash;

        emit CertificateIssued(_fileHash, _certificateId, _studentName);
    }

    /// @notice Verify by file hash (primary method — used after hashing uploaded file)
    function verifyCertificate(string memory _fileHash) public view returns (
        bool isValid,
        string memory certificateId,
        string memory studentName,
        string memory course,
        string memory ipfsHash,
        uint256 timestamp
    ) {
        Certificate memory cert = certsByHash[_fileHash];
        if (cert.isValid) {
            return (true, cert.certificateId, cert.studentName, cert.course, cert.ipfsHash, cert.timestamp);
        }
        return (false, "", "", "", "", 0);
    }

    /// @notice Get stored file hash by certificate ID (used for QR code verification)
    function getHashByCertId(string memory _certificateId) public view returns (string memory fileHash) {
        return hashByCertId[_certificateId];
    }

    /// @notice Get full certificate details by certificate ID (for QR display)
    function getCertificateById(string memory _certificateId) public view returns (
        bool isValid,
        string memory certificateId,
        string memory studentName,
        string memory course,
        string memory ipfsHash,
        string memory fileHash,
        uint256 timestamp
    ) {
        string memory hash = hashByCertId[_certificateId];
        if (bytes(hash).length == 0) {
            return (false, "", "", "", "", "", 0);
        }
        Certificate memory cert = certsByHash[hash];
        return (cert.isValid, cert.certificateId, cert.studentName, cert.course, cert.ipfsHash, cert.fileHash, cert.timestamp);
    }
}

FILE: backend\deploy.js

EXPLANATION: Deploys the compiled smart contract ABI/BIN to Ganache using ethers.js and the default private key, then sets the address in the .env file.

const fs = require('fs');
const path = require('path');
const ethers = require('ethers');

async function deploy() {
    const compiledDir = path.join(__dirname, '../smart_contracts/compiled');
    const files = fs.readdirSync(compiledDir);

    // Find the ABI and BIN files
    const abiFile = files.find(f => f.endsWith('.abi'));
    const binFile = files.find(f => f.endsWith('.bin'));

    if (!abiFile || !binFile) {
        console.error('ERROR: Could not find compiled .abi or .bin files in', compiledDir);
        console.error('Files found:', files);
        process.exit(1);
    }

    console.log('Using compiled files:', abiFile, binFile);

    const abi = JSON.parse(fs.readFileSync(path.join(compiledDir, abiFile), 'utf8'));
    const bytecode = fs.readFileSync(path.join(compiledDir, binFile), 'utf8').trim();

    // Connect to local Ganache (running on 8545)
    const provider = new ethers.JsonRpcProvider('http://127.0.0.1:8545');

    // Use Ganache's default first account private key (from the mnemonic "test test test...")
    const privateKey = '0xac0974bec39a17e36ba4a6b4d238ff944bacb478cbed5efcae784d7bf4f2ff80';
    const wallet = new ethers.Wallet(privateKey, provider);

    console.log('Deploying with wallet:', wallet.address);

    const factory = new ethers.ContractFactory(abi, bytecode, wallet);
    console.log('Deploying CertificateRegistry...');
    
    const contract = await factory.deploy();
    await contract.waitForDeployment();
    
    const address = await contract.getAddress();
    console.log('SUCCESS! CertificateRegistry deployed at:', address);

    // Write address to .env file
    const envPath = path.join(__dirname, '.env');
    let envContent = fs.existsSync(envPath) ? fs.readFileSync(envPath, 'utf8') : '';

    if (envContent.includes('CONTRACT_ADDRESS=')) {
        envContent = envContent.replace(/CONTRACT_ADDRESS=.*/, `CONTRACT_ADDRESS=${address}`);
    } else {
        envContent += `\nCONTRACT_ADDRESS=${address}`;
    }

    fs.writeFileSync(envPath, envContent);
    console.log('.env updated with CONTRACT_ADDRESS =', address);
    return address;
}

deploy().catch(err => {
    console.error('Deploy failed:', err.message);
    process.exit(1);
});

FILE: backend\server.js

EXPLANATION: Express backend entry point. Has endpoints for /api/admin/prepare-issue, /api/verify, and /api/verify-qr. Imports other services to process hashing and OCR.

======================================================================


require('dotenv').config();
const express = require('express');
const cors = require('cors');
const multer = require('multer');
const QRCode = require('qrcode');
const { pinFileToIPFS, getFileHash } = require('./mockIpfs');
const { extractCertificateId } = require('./ocrService');
const { issueOnBlockchain, verifyByFileHash, getHashByCertId, getCertificateById, abi } = require('./smartContractService');

const app = express();
app.use(cors());
app.use(express.json());

const upload = multer({ storage: multer.memoryStorage() });

// ─────────────────────────────────────────────────────────────────────────────
// ALGORITHM: Certificate Prep (File verification -> Hash -> IPFS -> QR)
// ─────────────────────────────────────────────────────────────────────────────
app.post('/api/admin/prepare-issue', upload.single('certificate'), async (req, res) => {
    try {
        if (!req.file) return res.status(400).json({ success: false, error: 'No file provided' });

        const { studentName, course, certificateId } = req.body;
        if (!studentName || !course || !certificateId) {
            return res.status(400).json({ success: false, error: 'studentName, course, and certificateId are required' });
        }

        // Generate SHA-256 hash of the certificate file
        const { cid, fileHash } = await pinFileToIPFS(req.file.buffer);
        console.log(`[Prepare Issue] SHA-256 hash = ${fileHash}, IPFS CID = ${cid}`);

        // Generate QR Code → encodes the Certificate ID for fast retrieval
        const qrData = `certid:${certificateId}`;
        const qrCodeDataUrl = await QRCode.toDataURL(qrData, { width: 200, margin: 1 });
        console.log(`[Prepare Issue] QR code generated for cert ID = ${certificateId}`);

        res.json({
            success: true,
            fileHash,
            ipfsHash: cid,
            qrCode: qrCodeDataUrl,
            message: 'File processing complete. Ready for MetaMask signature.'
        });

    } catch (error) {
        console.error('[Prepare Issue] Error:', error.message);
        res.status(500).json({ success: false, error: error.message });
    }
});

// ─────────────────────────────────────────────────────────────────────────────
// ALGORITHM: Certificate Verification (Image Upload)
// Step 1: User uploads certificate image
// Step 2: Apply OCR using Tesseract → extract Certificate ID
// Step 3: Generate SHA-256 hash of uploaded file
// Step 4: Fetch stored hash from Blockchain using Certificate ID
// Step 5: Compare Generated Hash == Stored Hash
//         → VALID if match, FAKE if not
// Step 6: Display result
// ─────────────────────────────────────────────────────────────────────────────
app.post('/api/verify', upload.single('certificate'), async (req, res) => {
    try {
        if (!req.file) return res.status(400).json({ success: false, error: 'No file provided' });

        // STEP 3: Generate SHA-256 hash of uploaded file
        const generatedHash = getFileHash(req.file.buffer);
        console.log(`[Verify] Step 3: Generated hash = ${generatedHash.substring(0, 16)}...`);

        let extractedId = null;
        let ocrText = '';

        // STEP 2: Apply OCR for images to extract Certificate ID
        if (req.file.mimetype.startsWith('image/')) {
            try {
                console.log('[Verify] Step 2: Running OCR...');
                const ocrResult = await extractCertificateId(req.file.buffer);
                extractedId = ocrResult.extractedId;
                ocrText = ocrResult.text ? ocrResult.text.trim().substring(0, 300) : '';
                console.log(`[Verify] Step 2: OCR extracted ID = "${extractedId}"`);
            } catch (ocrErr) {
                console.warn('[Verify] OCR failed (non-critical):', ocrErr.message);
            }
        }

        // STEP 4 & 5: Determine result
        // Primary path: look up blockchain by GENERATED HASH (most reliable — exact file match)
        const blockchainData = await verifyByFileHash(generatedHash);

        let isAuthentic = false;
        let reason = '';
        let ocrMatch = 'N/A';
        let certData = null;

        if (blockchainData.isValid) {
            // Hash matched on chain → VALID
            isAuthentic = true;
            certData = blockchainData;
            console.log('[Verify] Step 5: Hash matched blockchain → VALID');

            // Also cross-check OCR ID vs stored ID
            if (extractedId) {
                ocrMatch = extractedId === blockchainData.certificateId
                    ? `✓ OCR ID matched (${extractedId})`
                    : `⚠ OCR ID mismatch — OCR: "${extractedId}", Chain: "${blockchainData.certificateId}"`;
            }
        } else {
            // Hash not found — try lookup by OCR-extracted Certificate ID
            // Step 4: Fetch stored HASH from blockchain using extracted Certificate ID
            if (extractedId) {
                console.log(`[Verify] Step 4: Hash not found, trying lookup by OCR cert ID = "${extractedId}"`);
                const storedHash = await getHashByCertId(extractedId);

                if (storedHash && storedHash.length > 0) {
                    // Step 5: Compare generated hash vs stored hash
                    if (generatedHash === storedHash) {
                        // Should not normally reach here (already checked above), but handles edge cases
                        isAuthentic = true;
                        const certById = await getCertificateById(extractedId);
                        certData = certById;
                        ocrMatch = `✓ OCR ID matched and hash verified (${extractedId})`;
                        console.log('[Verify] Step 5: Hash matched via cert ID → VALID');
                    } else {
                        isAuthentic = false;
                        reason = `Certificate ID "${extractedId}" exists on blockchain, but the FILE HAS BEEN TAMPERED — hash does not match.`;
                        ocrMatch = `✗ Hash mismatch: generated vs stored`;
                        console.log('[Verify] Step 5: Cert ID found but hash mismatch → FAKE/TAMPERED');
                    }
                } else {
                    isAuthentic = false;
                    reason = extractedId
                        ? `Certificate ID "${extractedId}" not found on blockchain.`
                        : 'File hash not found on blockchain — certificate was not issued or has been tampered with.';
                    console.log('[Verify] Step 5: Not found on blockchain → FAKE');
                }
            } else {
                isAuthentic = false;
                reason = 'File hash not found on blockchain.';
            }
        }

        res.json({
            success: true,
            isAuthentic,
            reason: isAuthentic ? null : reason,
            message: isAuthentic
                ? '✅ Certificate is VALID — confirmed on the blockchain.'
                : '❌ Certificate is FAKE or TAMPERED.',
            generatedHash,
            extractedId,
            ocrText,
            ocrMatch,
            certificateData: certData
        });

    } catch (error) {
        console.error('[Verify] Error:', error.message);
        res.status(500).json({ success: false, error: error.message });
    }
});

// ─────────────────────────────────────────────────────────────────────────────
// ALGORITHM: QR Code Verification
// Step 1: Scan QR Code → extract Certificate ID
// Step 2: Fetch certificate details from Blockchain using Certificate ID
// Step 3: Display Student Name, Course, Status
// ─────────────────────────────────────────────────────────────────────────────
app.get('/api/verify-qr/:certificateId', async (req, res) => {
    try {
        const { certificateId } = req.params;
        console.log(`[QR Verify] Looking up cert ID = "${certificateId}"`);

        const certData = await getCertificateById(certificateId);

        if (!certData.isValid) {
            return res.json({
                success: true,
                isAuthentic: false,
                message: `❌ Certificate ID "${certificateId}" not found on blockchain.`
            });
        }

        res.json({
            success: true,
            isAuthentic: true,
            message: '✅ Certificate is VALID — confirmed on the blockchain.',
            certificateData: certData
        });

    } catch (error) {
        console.error('[QR Verify] Error:', error.message);
        res.status(500).json({ success: false, error: error.message });
    }
});

// Health check
app.get('/api/health', (req, res) => {
    res.json({ ok: true, contractAddress: process.env.CONTRACT_ADDRESS || 'NOT SET' });
});

app.get('/api/contract-info', (req, res) => {
    res.json({
        success: true,
        address: process.env.CONTRACT_ADDRESS,
        abi: abi
    });
});

const PORT = process.env.PORT || 5000;
app.listen(PORT, () => {
    console.log(`✅ CertifyChain Backend on http://localhost:${PORT}`);
    console.log(`   Contract: ${process.env.CONTRACT_ADDRESS || 'NOT SET'}`);
});

require('dotenv').config();

// ── Issue Certificate on Blockchain ───────────────────────────────────────────

FILE: backend\ocrService.js

EXPLANATION: Uses Tesseract.js to read text from uploaded certificate images and attempts to extract a Certificate ID.

const Tesseract = require('tesseract.js');

/**
 * Extracts text from an image buffer and attempts to find a Certificate ID.
 * Since this is generic, we'll look for simple patterns like "ID: 12345" 
 * or just return the raw text if we can't cleanly parse it.
 */
async function extractCertificateId(imageBuffer) {
    try {
        const result = await Tesseract.recognize(imageBuffer, 'eng');
        const text = result.data.text;
        
        // Simple regex to find something like "Certificate ID: <ID>" or "ID: <ID>"
        // This regex looks for "id:", "id ", "ID -", etc., followed by alphanumeric string.
        const idRegex = /(?:certificate\s+id|id)[\s\:\-]+([A-Z0-9\-]+)/i;
        const match = text.match(idRegex);

        let extractedId = null;
        if (match && match[1]) {
            extractedId = match[1].trim();
        }

        return { text, extractedId };
    } catch (error) {
        console.error("OCR Error:", error);
        throw error;
    }
}

module.exports = { extractCertificateId };

FILE: backend\mockIpfs.js

EXPLANATION: Generates a true SHA-256 hash using crypto.createHash and mocks IPFS CID generation.

======================================================================


const fs = require('fs');
const path = require('path');
const crypto = require('crypto');

const IPFS_DIR = path.join(__dirname, 'ipfs-data');

if (!fs.existsSync(IPFS_DIR)) {
  fs.mkdirSync(IPFS_DIR, { recursive: true });
}

/**
 * Mocks pinning a file to IPFS. Returns a pseudo-CID based on the SHA-256 hash.
 */
function pinFileToIPFS(buffer) {
  return new Promise((resolve) => {
    // Generate a pseudo-CID using sha256. In real IPFS, it uses multihash (e.g. Qm...)
    const hash = crypto.createHash('sha256').update(buffer).digest('hex');
    const cid = `QmMockIPFS${hash.substring(0, 36)}`; 
    
    // Save to local directory to mock decentralized storage
    const filePath = path.join(IPFS_DIR, cid);
    fs.writeFileSync(filePath, buffer);
    
    // Note: We also need the raw file hash for the smart contract mapping 
    // (since it maps SHA-256 fileHash -> Certificate details)
    resolve({ cid, fileHash: hash });
  });
}

function getFileHash(buffer) {
  return crypto.createHash('sha256').update(buffer).digest('hex');
}

module.exports = { pinFileToIPFS, getFileHash };

function WalletButton({ onAddressChange }) {
    const [address, setAddress] = useState(null);
    const [showModal, setShowModal] = useState(false);

    useEffect(() => {
        if (window.ethereum) {
            window.ethereum.request({ method: 'eth_accounts' })
                .then(accs => { if (accs[0]) { setAddress(accs[0]); onAddressChange?.(accs[0]); } })
                .catch(() => {});
        }
    }, []);

    const handleConnected = useCallback((addr) => {
        setAddress(addr); onAddressChange?.(addr);
    }, [onAddressChange]);

    const disconnect = useCallback(() => { setAddress(null); onAddressChange?.(null); }, [onAddressChange]);

    if (address) {
        return (
            <button className="btn btn-outline btn-sm" onClick={disconnect} title="Disconnect"
                style={{ fontFamily: 'monospace', display: 'flex', alignItems: 'center', gap: '0.45rem' }}>
                <span style={{ width: 8, height: 8, borderRadius: '50%', background: '#22c55e', boxShadow: '0 0 6px #22c55e', display: 'inline-block' }} />
                {address.slice(0, 6)}…{address.slice(-4)}
            </button>
        );
    }

    return (
        <>
            <button className="btn btn-primary btn-sm" onClick={() => setShowModal(true)}>
                🔗 Connect Wallet
            </button>
            {showModal && <WalletModal onClose={() => setShowModal(false)} onConnected={handleConnected} />}

/* ── Blockchain Network Canvas ─────────────────────────────────────────── */
const BlockchainCanvas = () => {
    const canvasRef = useRef(null);
    const rafRef = useRef(null);

    useEffect(() => {
        const canvas = canvasRef.current;
        if (!canvas) return;
        const ctx = canvas.getContext('2d');

        const resize = () => { canvas.width = window.innerWidth; canvas.height = window.innerHeight; };
        resize();
        window.addEventListener('resize', resize);

        const NODE_COUNT = 40;
        const nodes = Array.from({ length: NODE_COUNT }, () => ({
            x: Math.random() * canvas.width,
            y: Math.random() * canvas.height,
            vx: (Math.random() - 0.5) * 0.5,
            vy: (Math.random() - 0.5) * 0.5,
            r: 2 + Math.random() * 3,
            pulse: Math.random() * Math.PI * 2,
            pulseSpeed: 0.015 + Math.random() * 0.02,
            color: Math.random() < 0.6 ? '#00f0ff' : (Math.random() < 0.5 ? '#b44fff' : '#00ff88'),
        }));

        // Data packets on edges
        const packets = Array.from({ length: 12 }, () => ({
            fromIdx: Math.floor(Math.random() * NODE_COUNT),
            toIdx: Math.floor(Math.random() * NODE_COUNT),
            progress: Math.random(),
            speed: 0.003 + Math.random() * 0.005,
            color: Math.random() < 0.5 ? '#00f0ff' : '#b44fff',
        }));

        // Floating hash labels
        const LABELS = ['0x4f3a…', 'TX✓', '#sha256', 'BLOCK', '⛓', 'CERT-ID', 'HASH', '0xA1B2…', 'VALID', 'SIGN'];
        const floaters = Array.from({ length: 14 }, (_, i) => ({
            text: LABELS[i % LABELS.length],
            x: Math.random() * canvas.width,
            y: Math.random() * canvas.height,
            vy: -(0.15 + Math.random() * 0.25),
            opacity: 0.08 + Math.random() * 0.18,
            size: 10 + Math.random() * 5,
            color: Math.random() < 0.5 ? '#00f0ff' : '#b44fff',
        }));

        const CONNECT_DIST = 160;

        const animate = () => {
            ctx.clearRect(0, 0, canvas.width, canvas.height);

            // Move nodes
            for (const n of nodes) {
                n.x += n.vx; n.y += n.vy; n.pulse += n.pulseSpeed;
                if (n.x < 0 || n.x > canvas.width) n.vx *= -1;
                if (n.y < 0 || n.y > canvas.height) n.vy *= -1;
            }

            // Draw edges
            for (let i = 0; i < nodes.length; i++) {
                for (let j = i + 1; j < nodes.length; j++) {
                    const dx = nodes[i].x - nodes[j].x;
                    const dy = nodes[i].y - nodes[j].y;
                    const dist = Math.sqrt(dx * dx + dy * dy);
                    if (dist < CONNECT_DIST) {
                        const alpha = (1 - dist / CONNECT_DIST) * 0.35;
                        const grad = ctx.createLinearGradient(nodes[i].x, nodes[i].y, nodes[j].x, nodes[j].y);
                        grad.addColorStop(0, `rgba(0,240,255,${alpha})`);
                        grad.addColorStop(1, `rgba(180,79,255,${alpha})`);
                        ctx.beginPath();
                        ctx.moveTo(nodes[i].x, nodes[i].y);
                        ctx.lineTo(nodes[j].x, nodes[j].y);
                        ctx.strokeStyle = grad;
                        ctx.lineWidth = 1;
                        ctx.stroke();
                    }
                }
            }

            // Draw data packets
            for (const pkt of packets) {
                pkt.progress += pkt.speed;
                if (pkt.progress >= 1) {
                    pkt.progress = 0;
                    pkt.fromIdx = pkt.toIdx;
                    pkt.toIdx = Math.floor(Math.random() * NODE_COUNT);
                }
                const from = nodes[pkt.fromIdx];
                const to = nodes[pkt.toIdx];
                const px = from.x + (to.x - from.x) * pkt.progress;
                const py = from.y + (to.y - from.y) * pkt.progress;
                ctx.beginPath();
                ctx.arc(px, py, 3, 0, Math.PI * 2);
                ctx.fillStyle = pkt.color;
                ctx.shadowColor = pkt.color;
                ctx.shadowBlur = 12;
                ctx.fill();
                ctx.shadowBlur = 0;
            }

            // Draw nodes
            for (const n of nodes) {
                const glow = 0.5 + 0.5 * Math.sin(n.pulse);
                // Outer ring
                ctx.beginPath(); ctx.arc(n.x, n.y, n.r + 5 * glow, 0, Math.PI * 2);
                const ringAlpha = 0.08 * glow;
                ctx.fillStyle = n.color.replace(')', `,${ringAlpha})`).replace('rgb', 'rgba');
                // Simpler approach:
                ctx.globalAlpha = 0.08 * glow;
                ctx.fillStyle = n.color;
                ctx.fill();
                ctx.globalAlpha = 1;

                // Core dot
                ctx.beginPath(); ctx.arc(n.x, n.y, n.r, 0, Math.PI * 2);
                ctx.fillStyle = n.color;
                ctx.shadowColor = n.color;
                ctx.shadowBlur = 10 + 10 * glow;
                ctx.fill();
                ctx.shadowBlur = 0;
            }

            // Floating hash labels
            for (const f of floaters) {
                f.y += f.vy;
                if (f.y < -20) { f.y = canvas.height + 20; f.x = Math.random() * canvas.width; }
                ctx.save();
                ctx.globalAlpha = f.opacity;
                ctx.fillStyle = f.color;
                ctx.font = `${f.size}px 'JetBrains Mono', 'Courier New', monospace`;
                ctx.shadowColor = f.color;
                ctx.shadowBlur = 6;
                ctx.fillText(f.text, f.x, f.y);
                ctx.restore();
            }

            rafRef.current = requestAnimationFrame(animate);
        };

        animate();
        return () => {
            window.removeEventListener('resize', resize);
            if (rafRef.current) cancelAnimationFrame(rafRef.current);
        };
    }, []);

    return <canvas ref={canvasRef} style={{ position: 'fixed', inset: 0, zIndex: -2, pointerEvents: 'none' }} />;
};

/* ── Mouse spark trail ─────────────────────────────────────────────────── */
const MouseTrail = () => {
    const canvasRef = useRef(null);
    const particles = useRef([]);
    const rafRef = useRef(null);

    useEffect(() => {
        const canvas = canvasRef.current;
        if (!canvas) return;
        const ctx = canvas.getContext('2d');
        const resize = () => { canvas.width = window.innerWidth; canvas.height = window.innerHeight; };
        resize();
        window.addEventListener('resize', resize);

        const COLORS = ['#00f0ff', '#b44fff', '#00ff88', '#4f46e5', '#00f0ff'];

        const onMove = (e) => {
            for (let i = 0; i < 4; i++) {
                particles.current.push({
                    x: e.clientX, y: e.clientY,
                    vx: (Math.random() - 0.5) * 3,
                    vy: (Math.random() - 0.5) * 3 - 0.5,
                    size: 1.5 + Math.random() * 5,
                    opacity: 1,
                    decay: 0.045 + Math.random() * 0.035,
                    color: COLORS[Math.floor(Math.random() * COLORS.length)],
                    gravity: 0.06,
                });
            }
        };
        window.addEventListener('mousemove', onMove);

        const animate = () => {
            ctx.clearRect(0, 0, canvas.width, canvas.height);
            particles.current = particles.current.filter(p => p.opacity > 0.01);
            for (const p of particles.current) {
                ctx.save();
                ctx.globalAlpha = p.opacity;
                ctx.fillStyle = p.color;
                ctx.shadowColor = p.color;
                ctx.shadowBlur = 10;
                ctx.beginPath();
                ctx.arc(p.x, p.y, p.size, 0, Math.PI * 2);
                ctx.fill();
                ctx.restore();
                p.x += p.vx; p.y += p.vy;
                p.vy += p.gravity; p.opacity -= p.decay;
            }
            rafRef.current = requestAnimationFrame(animate);
        };
        animate();

        return () => {
            window.removeEventListener('resize', resize);
            window.removeEventListener('mousemove', onMove);
            if (rafRef.current) cancelAnimationFrame(rafRef.current);
        };
    }, []);

    return <canvas ref={canvasRef} style={{ position: 'fixed', inset: 0, zIndex: 9999, pointerEvents: 'none' }} />;
};

        await animateStep(0); await animateStep(1); await animateStep(2);
        try {
            const resp = await fetch(`http://localhost:5000/api/verify-qr/${encodeURIComponent(certId.trim())}`);
            const json = await resp.json();
            await animateStep(3); await animateStep(4);
            if (json.success) { await animateStep(5); setResult(json); }
            else setError(json.error || 'Verification error');
        } catch { setError('Cannot connect to backend. Is the server running?'); }
        finally { setLoading(false); }
    };

    const getStepState = (index) => {
        if (completedSteps.includes(index)) return 'completed';
        if (currentStep === index) return 'active';
        return '';
    };

    const activeSteps = tab === 0 ? IMAGE_ALGO_STEPS : QR_ALGO_STEPS;
    const isValid = result?.isAuthentic;
    const formatFileSize = (bytes) => {
        if (bytes < 1024) return `${bytes} B`;
        if (bytes < 1048576) return `${(bytes / 1024).toFixed(1)} KB`;
        return `${(bytes / 1048576).toFixed(1)} MB`;
    };
    const fileIcon = file?.type === 'application/pdf' ? '📄' : '🖼️';

    return (
        <div className="verify-page page-enter">
            <div className="page-header" style={{ textAlign: 'center', marginBottom: '1.5rem' }}>
                <h2 className="gradient-text">Verify Certificate</h2>
                <p>Upload an image for OCR+blockchain check, or enter a Certificate ID</p>
            </div>

            {/* Algorithm Stepper */}
            <div className="algo-stepper" key={tab}>
                {activeSteps.map((step, i) => (
                    <React.Fragment key={i}>
                        <div className={`algo-step ${getStepState(i)}`}>
                            <div className="step-number">{completedSteps.includes(i) ? '✓' : step.num}</div>
                            <span className="step-label">{step.label}</span>
                        </div>
                        {i < activeSteps.length - 1 && (
                            <div className={`step-connector ${completedSteps.includes(i) ? 'completed' : ''}`} />
                        )}
                    </React.Fragment>
                ))}
            </div>

            <div className="glass-panel">
                <div style={{ textAlign: 'center', marginBottom: '1.5rem' }}>
                    <h3 style={{ marginBottom: '0.5rem' }}>Verify Certificate Authenticity</h3>
                    <p style={{ fontSize: '0.9rem' }}>
                        Upload the certificate image/PDF for OCR+blockchain check, or enter a Certificate ID for instant lookup.
                    </p>
                </div>

                {/* Tabs */}
                <div className="tab-bar">
                    {TABS.map((t, i) => (
                        <button key={i} className={`tab-btn ${tab === i ? 'active' : ''}`}
                            onClick={() => { setTab(i); reset(); setFile(null); setCertId(''); setIsDragOver(false); }}>
                            {t}
                        </button>
                    ))}
                </div>

                {/* Tab 1: File Upload */}
                {tab === 0 && (
                    <form onSubmit={handleFileVerify} style={{ marginTop: '1rem' }}>
                        <div
                            className={`dropzone ${isDragOver ? 'drag-over' : ''} ${file && !isDragOver ? 'has-file' : ''}`}
                            onClick={() => fileInputRef.current?.click()}
                            onDragEnter={handleDragEnter} onDragLeave={handleDragLeave}
                            onDragOver={handleDragOver} onDrop={handleDrop}
                        >
                            <input ref={fileInputRef} type="file" id="verify-file" accept="image/*,application/pdf"
                                style={{ display: 'none' }} onChange={e => handleFileSelect(e.target.files[0])} />
                            {isDragOver ? (
                                <div className="dropzone-drag-content">
                                    <span className="drop-icon" style={{ fontSize: '3.5rem' }}>🎯</span>
                                    <h3 style={{ color: '#00f0ff' }}>Drop it here!</h3>
                                    <p>Release to upload</p>
                                </div>
                            ) : file ? (
                                <div className="dropzone-file-content">
                                    <span className="drop-icon">{fileIcon}</span>
                                    <h3 style={{ color: '#00ff88' }}>{file.name}</h3>
                                    <p style={{ color: '#5a7a99' }}>{formatFileSize(file.size)} — Click or drag to change</p>
                                    <div className="file-type-badge">{file.type === 'application/pdf' ? 'PDF' : 'IMAGE'}</div>
                                </div>
                            ) : (
                                <div className="dropzone-empty-content">
                                    <span className="drop-icon">📤</span>
                                    <h3>Drag & Drop or Click to Browse</h3>
                                    <p>Supports PNG, JPG, WEBP, PDF</p>
                                    <div className="dropzone-hint">
                                        <span>🖱️ Click</span><span>or</span><span>📂 Drag file here</span>
                                    </div>
                                </div>
                            )}
                        </div>
                        <div style={{ textAlign: 'center', marginTop: '1.5rem' }}>
                            <button type="submit" className="btn btn-primary btn-lg" disabled={loading} style={{ minWidth: 240 }}>
                                {loading ? <><span className="loader"></span>Verifying...</> : '🔍 Verify Now'}
                            </button>
                        </div>
                    </form>
                )}

                {/* Tab 2: Certificate ID */}
                {tab === 1 && (
                    <form onSubmit={handleQRVerify} style={{ marginTop: '1rem' }}>
                        <div style={{
                            textAlign: 'center', marginBottom: '1.5rem', padding: '1.5rem',
                            background: 'rgba(0,240,255,0.03)', border: '1px solid rgba(0,240,255,0.10)', borderRadius: 12,
                        }}>
                            <div style={{ fontSize: '3rem', marginBottom: '0.75rem' }}>📷</div>
                            <p style={{ marginBottom: '1.5rem', color: '#5a7a99', fontSize: '0.9rem' }}>
                                Scan the QR code on the certificate, then paste the Certificate ID below. Or enter it manually.
                            </p>
                            <div className="form-group" style={{ textAlign: 'left', maxWidth: 420, margin: '0 auto' }}>
                                <label className="form-label">Certificate ID</label>
                                <input className="form-input" type="text" placeholder="e.g. CERT-2024-0042"
                                    value={certId} onChange={e => { setCertId(e.target.value); reset(); }} required />
                            </div>
                        </div>
                        <div style={{ textAlign: 'center' }}>
                            <button type="submit" className="btn btn-primary btn-lg" disabled={loading} style={{ minWidth: 240 }}>
                                {loading ? <><span className="loader"></span>Looking up...</> : '🔗 Fetch from Blockchain'}
                            </button>
                        </div>
                    </form>
                )}

                {/* Error */}
                {error && (
                    <div className="result-banner result-fake" style={{ marginTop: '1.5rem' }}>
                        <div style={{ display: 'flex', alignItems: 'center', gap: '0.75rem' }}>
                            <span style={{ fontSize: '1.5rem' }}>❌</span>
                            <div>
                                <strong style={{ color: '#ff3c3c' }}>Verification Error</strong>
                                <p style={{ margin: 0, color: '#ff3c3c', fontSize: '0.9rem' }}>{error}</p>
                            </div>
                        </div>
                    </div>
                )}

                {/* Result */}
                {result && (
                    <div className={`result-banner ${isValid ? 'result-valid' : 'result-fake'}`} style={{ marginTop: '2rem' }}>
                        <div style={{ display: 'flex', alignItems: 'center', gap: '1rem', marginBottom: '1.25rem', flexWrap: 'wrap' }}>
                            <span style={{
                                display: 'inline-flex', alignItems: 'center', gap: '0.4rem',
                                padding: '0.35rem 0.9rem', borderRadius: 50,
                                fontSize: '0.75rem', fontWeight: 800, letterSpacing: '0.05em',
                                background: isValid ? 'rgba(0,255,136,0.15)' : 'rgba(255,60,60,0.15)',
                                color: isValid ? '#00ff88' : '#ff3c3c',
                                border: `1px solid ${isValid ? 'rgba(0,255,136,0.4)' : 'rgba(255,60,60,0.4)'}`,
                            }}>
                                {isValid ? '✅ VALID' : '❌ FAKE / TAMPERED'}
                            </span>
                            <p style={{ color: isValid ? '#00ff88' : '#ff3c3c', margin: 0, fontWeight: 500 }}>{result.message}</p>
                        </div>

                        {!isValid && result.reason && (
                            <p style={{ color: '#ff3c3c', marginBottom: '1rem', fontSize: '0.9rem' }}>{result.reason}</p>
                        )}

                        {isValid && result.certificateData && (
                            <div style={{
                                background: 'rgba(0,240,255,0.04)',
                                backdropFilter: 'blur(16px)',
                                borderRadius: 14, padding: '1.5rem', marginBottom: '1.5rem',
                                border: '1px solid rgba(0,240,255,0.15)',
                                boxShadow: '0 0 30px rgba(0,240,255,0.06)',
                                position: 'relative', overflow: 'hidden',
                            }}>
                                {/* HUD corner brackets */}
                                <div style={{ position: 'absolute', top: 8, left: 8, width: 12, height: 12, borderTop: '2px solid rgba(0,240,255,0.6)', borderLeft: '2px solid rgba(0,240,255,0.6)', borderRadius: '2px 0 0 0' }} />
                                <div style={{ position: 'absolute', bottom: 8, right: 8, width: 12, height: 12, borderBottom: '2px solid rgba(0,240,255,0.6)', borderRight: '2px solid rgba(0,240,255,0.6)', borderRadius: '0 0 2px 0' }} />
                                <div className="data-table">
                                    <div className="data-row">
                                        <span>Student Name</span>
                                        <strong style={{ color: '#e8f4ff' }}>{result.certificateData.studentName}</strong>
                                    </div>
                                    <div className="data-row">
                                        <span>Course</span>
                                        <strong style={{ color: '#e8f4ff' }}>{result.certificateData.course}</strong>
                                    </div>
                                    <div className="data-row">
                                        <span>Certificate ID</span>
                                        <code style={{ color: '#00f0ff', fontWeight: 700, fontFamily: "'JetBrains Mono',monospace", textShadow: '0 0 8px rgba(0,240,255,0.5)', fontSize: '0.85rem' }}>
                                            {result.certificateData.certificateId}
                                        </code>
                                    </div>
                                    <div className="data-row">
                                        <span>IPFS Storage</span>
                                        <code className="hash">{result.certificateData.ipfsHash}</code>
                                    </div>
                                    <div className="data-row">
                                        <span>Status</span>
                                        <strong style={{ color: '#00ff88', textShadow: '0 0 8px rgba(0,255,136,0.4)' }}>✅ Authenticated on Blockchain</strong>
                                    </div>
                                    <div className="data-row">
                                        <span>Issued At</span>
                                        <strong style={{ color: '#a8c4e0', fontFamily: "'JetBrains Mono',monospace", fontSize: '0.82rem' }}>
                                            {result.certificateData.timestamp
                                                ? new Date(Number(result.certificateData.timestamp) * 1000).toLocaleString()
                                                : 'N/A'}
                                        </strong>
                                    </div>
                                </div>
                            </div>
                        )}

                        {result.generatedHash && (
                            <div className="processing-log">
                                <div className="log-entry success">✓ OCR scan completed</div>
                                {result.extractedId && <div className="log-entry success">✓ Extracted ID: {result.extractedId}</div>}
                                <div className="log-entry success">✓ SHA-256 hash: {result.generatedHash.substring(0, 28)}...</div>
                                <div className="log-entry success">✓ Blockchain query complete</div>
                                <div className="log-entry success">
                                    ✓ Result: {isValid ? 'Hash MATCHED — Certificate is VALID' : 'Hash MISMATCH — Certificate is FAKE'}
                                </div>
                                {result.ocrMatch && result.ocrMatch !== 'N/A' && (
                                    <div className="log-entry success">✓ OCR: {result.ocrMatch}</div>
                                )}
                            </div>
                        )}
                    </div>
                )}
            </div>
        </div>
    );
};

export default VerificationPortal;
